cmake_minimum_required(VERSION 3.22)
project(quicklook_softblur_v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Quick DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PLATFORM REQUIRED IMPORTED_TARGET epoxy wayland-server)
find_path(KWIN_INCLUDE_DIR effect/effect.h PATH_SUFFIXES kwin REQUIRED)
find_library(KWIN_LIBRARY kwin REQUIRED)
file(READ "${KWIN_INCLUDE_DIR}/config-kwin.h" KWIN_CONFIG)
if(NOT KWIN_CONFIG MATCHES "KWIN_PLUGIN_VERSION_STRING \"6\\.7\\.5\"")
    message(FATAL_ERROR "El efecto suave requiere exactamente KWin 6.7.5; el visor Qt funciona sin este módulo")
endif()
find_path(KF_CORE_INCLUDE KPluginFactory PATH_SUFFIXES KF6/KCoreAddons REQUIRED)
find_path(KF_CONFIG_INCLUDE KSharedConfig PATH_SUFFIXES KF6/KConfigCore REQUIRED)
find_path(KF_CONFIG_VERSION_INCLUDE kconfig_version.h PATH_SUFFIXES KF6/KConfig REQUIRED)
find_path(KF_WINDOW_INCLUDE KWindowSystem PATH_SUFFIXES KF6/KWindowSystem REQUIRED)
add_library(quicklook_softblur_v2 MODULE softblur.cpp shaders.qrc)
target_include_directories(quicklook_softblur_v2 PRIVATE ${KWIN_INCLUDE_DIR} ${KF_CORE_INCLUDE}
    ${KF_CONFIG_INCLUDE} ${KF_CONFIG_VERSION_INCLUDE} ${KF_WINDOW_INCLUDE})
target_link_libraries(quicklook_softblur_v2 PRIVATE ${KWIN_LIBRARY} Qt6::Widgets Qt6::Quick Qt6::DBus
    KF6CoreAddons KF6ConfigCore KF6WindowSystem PkgConfig::PLATFORM)
target_compile_options(quicklook_softblur_v2 PRIVATE -Wall -Wextra -Werror)
set_target_properties(quicklook_softblur_v2 PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/kwin/effects/plugins")
