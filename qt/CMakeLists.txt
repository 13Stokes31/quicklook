cmake_minimum_required(VERSION 3.22)
project(quicklook_qt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WINDOWSYSTEM REQUIRED IMPORTED_TARGET KF6WindowSystem)
set(QUICKLOOK_CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../target/release" CACHE PATH "Directorio del núcleo Rust")
find_library(QUICKLOOK_CORE quicklook_core PATHS "${QUICKLOOK_CORE_DIR}" NO_DEFAULT_PATH REQUIRED)
add_executable(quicklook-qt main.cpp)
target_compile_options(quicklook-qt PRIVATE -Wall -Wextra -Werror)
target_link_libraries(quicklook-qt PRIVATE ${QUICKLOOK_CORE} Qt6::Widgets Qt6::DBus PkgConfig::WINDOWSYSTEM dl pthread m)
install(PROGRAMS "$<TARGET_FILE:quicklook-qt>" DESTINATION bin RENAME quicklook)

option(QUICKLOOK_PDF "Compilar el módulo PDF si Qt PDF está disponible" ON)
if(QUICKLOOK_PDF)
    find_package(Qt6 QUIET COMPONENTS Pdf PdfWidgets)
    if(TARGET Qt6::PdfWidgets)
        add_library(quicklook-pdf MODULE pdfview.cpp)
        set_target_properties(quicklook-pdf PROPERTIES PREFIX "")
        target_compile_options(quicklook-pdf PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(quicklook-pdf PRIVATE Qt6::PdfWidgets)
        install(TARGETS quicklook-pdf LIBRARY DESTINATION lib/quicklook)
    else()
        message(STATUS "Qt PDF no disponible: se compilan imágenes y texto sin módulo PDF")
    endif()
endif()

option(QUICKLOOK_BUILD_TESTS "Compilar las pruebas del lector de documentos" OFF)
if(QUICKLOOK_BUILD_TESTS)
    enable_testing()
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(check-documents ../tests/check_documents.cpp)
    target_compile_options(check-documents PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(check-documents PRIVATE Qt6::Widgets Qt6::Test)
    if(TARGET quicklook-pdf)
        add_dependencies(check-documents quicklook-pdf)
        target_compile_definitions(check-documents PRIVATE QUICKLOOK_TEST_PDF=1)
    endif()
    add_test(NAME documents COMMAND check-documents -platform offscreen)
endif()
